void Renderer::Render(const Scene& scene)
{
    std::vector<Vector3f> framebuffer(scene.width * scene.height);

    float scale = tan(deg2rad(scene.fov * 0.5));
    float imageAspectRatio = scene.width / (float)scene.height;
    Vector3f eye_pos(-1, 5, 10);
    int m = 0;
    for (uint32_t j = 0; j < scene.height; ++j) {
        for (uint32_t i = 0; i < scene.width; ++i) {
            // generate primary ray direction
            float x = (2 * (i + 0.5) / (float)scene.width - 1) *
                imageAspectRatio * scale;
            float y = (1 - 2 * (j + 0.5) / (float)scene.height) * scale;
            // TODO: Find the x and y positions of the current pixel to get the
            // direction
            //  vector that passes through it.
            // Also, don't forget to multiply both of them with the variable
            // *scale*, and x (horizontal) variable with the *imageAspectRatio*
            Vector3f dir = normalize(Vector3f(x, y, -1));
            framebuffer[m++] = scene.castRay(Ray(eye_pos, dir), 0);
        }
        UpdateProgress(j / (float)scene.height);
    }
    UpdateProgress(1.f);

    // save framebuffer to file
    FILE* fp = fopen("binary.ppm", "wb");
    (void)fprintf(fp, "P6\n%d %d\n255\n", scene.width, scene.height);
    for (auto i = 0; i < scene.height * scene.width; ++i) {
        static unsigned char color[3];
        color[0] = (unsigned char)(255 * clamp(0, 1, framebuffer[i].x));
        color[1] = (unsigned char)(255 * clamp(0, 1, framebuffer[i].y));
        color[2] = (unsigned char)(255 * clamp(0, 1, framebuffer[i].z));
        fwrite(color, 1, 3, fp);
    }
    fclose(fp);
}

inline Intersection Triangle::getIntersection(Ray ray)
{
    Intersection inter;

    Vector3f e1 = v1 - v0;
    Vector3f e2 = v2 - v0;
    Vector3f s = ray.origin - v0;
    Vector3f s1 = crossProduct(ray.direction, e2);
    Vector3f s2 = crossProduct(s, e1);

    float denom = dotProduct(s1, e1);
    if (fabs(denom) < EPSILON)
        return inter;

    float t = dotProduct(s2, e2) / denom;
    float b1 = dotProduct(s1, s) / denom;
    float b2 = dotProduct(s2, ray.direction) / denom;

    float exp = -0.0001f;
    if (t > 0.f && b1 > exp && b2 > exp && (1 - b1 - b2) > exp) {
        inter.happened = true;
        inter.distance = t;
        inter.coords = ray.origin + t * ray.direction;
        inter.normal = normal;
        inter.obj = this;
        inter.m = m;
    }

    return inter;
}

inline bool Bounds3::IntersectP(const Ray& ray, const Vector3f& invDir,
    const std::array<int, 3>& dirIsNeg) const
{
    // invDir: ray direction(x,y,z), invDir=(1.0/x,1.0/y,1.0/z), use this because Multiply is faster that Division
    // dirIsNeg: ray direction(x,y,z), dirIsNeg=[int(x>0),int(y>0),int(z>0)], use this to simplify your logic
    // TODO test if ray bound intersects
    float xt0 = (pMin.x - ray.origin.x) * invDir.x;
    float xt1 = (pMax.x - ray.origin.x) * invDir.x;
    float yt0 = (pMin.y - ray.origin.y) * invDir.y;
    float yt1 = (pMax.y - ray.origin.y) * invDir.y;
    float zt0 = (pMin.z - ray.origin.z) * invDir.z;
    float zt1 = (pMax.z - ray.origin.z) * invDir.z;

    if (dirIsNeg[0])
        std::swap(xt0, xt1);
    if (dirIsNeg[1])
        std::swap(yt0, yt1);
    if (dirIsNeg[2])
        std::swap(zt0, zt1);

    float t0 = fmax(fmax(xt0, yt0), zt0);
    float t1 = fmin(fmin(xt1, yt1), zt1);

    if (t0 <= t1 && t1 >= 0) return true;
    return false;
}

Intersection BVHAccel::getIntersection(BVHBuildNode* node, const Ray& ray) const
{
    // TODO Traverse the BVH to find intersection
    Intersection isect;

    if (!node) return isect;
    Vector3f invDir(1.0f / ray.direction.x, 1.0f / ray.direction.y,
        1.0f / ray.direction.z);
    std::array<int, 3> dirIsNeg = { invDir.x < 0, invDir.y < 0, invDir.z < 0 };

    if (!node->bounds.IntersectP(ray, invDir, dirIsNeg)) {
        return isect;
    }
    else {
        if (node->left == nullptr && node->right == nullptr) {
            if (node->object != nullptr) {
                isect = node->object->getIntersection(ray);
            };
        }
        else {
            // Intersect with left and right child nodes
            Intersection isect1 = getIntersection(node->left, ray);
            Intersection isect2 = getIntersection(node->right, ray);
            if (isect1.happened || isect2.happened) {
                if (isect1.distance < isect2.distance) {
                    isect = isect1;
                }
                else {
                    isect = isect2;
                }
            }
        }
    }
    return isect;
}
